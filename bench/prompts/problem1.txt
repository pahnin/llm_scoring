function buildChatCommand(endpointUrl, apiKey, model, systemPrompt, memory, temperature) {
  var messages = [];

  var history = buildHistoryContext(conv, memory);

  messages = messages.concat(history);

  return {
    url: endpointUrl + "/v1/chat/completions",
    args: ["curl", "-X", "POST"]
  };
}

function buildHistoryContext(conv, memory) {
  var history = [];

  if (memory.summary) {
    history.push({
      role: "system",
      content: memory.summary
    });
  }

  history = history.concat(conv.messages.slice(-2));

  return history;
}
